cmake_minimum_required(VERSION 3.0.2)
project(v8_ndt_odom)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs std_msgs std_srvs pcl_ros ndt_omp)
find_package(PCL REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS roscpp sensor_msgs nav_msgs std_msgs std_srvs pcl_ros ndt_omp)
include_directories(include ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_executable(ndt_odom_node src/ndt_odom_node.cpp)
target_link_libraries(ndt_odom_node ${catkin_LIBRARIES} ${PCL_LIBRARIES})
set_target_properties(ndt_odom_node PROPERTIES COMPILE_FLAGS "-O2 -fopenmp")
add_executable(ndt_score_probe src/ndt_score_probe.cpp)
target_link_libraries(ndt_score_probe ${catkin_LIBRARIES} ${PCL_LIBRARIES})
set_target_properties(ndt_score_probe PROPERTIES COMPILE_FLAGS "-O2 -fopenmp")
if(CATKIN_ENABLE_TESTING)
  add_executable(ndt_history_test test/history_test.cpp)
  add_test(NAME ndt_history_test COMMAND ndt_history_test)
endif()
install(TARGETS ndt_odom_node ndt_score_probe
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
