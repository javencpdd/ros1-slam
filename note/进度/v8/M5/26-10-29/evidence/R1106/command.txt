# 0) 前置：容器内源码＝宿主 /home/jack/ysc-code/ros1-slam；宿主无 ROS。
#    v8_ndt_odom 依赖 geometry_msgs（新增 ~initialpose），先补 CMake/package.xml。

  source /opt/ros/noetic/setup.bash
  cd /opt/ros1-slam/ws
  catkin_make -DCATKIN_WHITELIST_PACKAGES="" --pkg v8_ndt_odom -j4'

# 1) 合成点云接口探针：11 case（accept/reject/reset/initialpose/relocalize）。
#    独立 roscore 11960；输出 ndt_probe_result.json 与 stdout。
docker exec ros1-slam-sim bash -lc '
  source /opt/ros/noetic/setup.bash
  source /opt/ros1-slam/ws/devel/setup.bash
  export ROS_MASTER_URI=http://127.0.0.1:11960 ROS_IP=127.0.0.1
  roscore -p 11960 >/tmp/a6_ndt_probe/roscore.log 2>&1 &
  sleep 3
  python3 -B /opt/ros1-slam/ws/src/90_platform/v8_ndt_odom/test/ros_monitor_probe.py \
    --binary /opt/ros1-slam/ws/devel/lib/v8_ndt_odom/ndt_odom_node \
    --out /tmp/a6_ndt_probe/ndt_probe_result.json'

# 2) Gazebo 活体：m1 世界，机器人静态在 (1.5,0)；NDT 用 /a6_ndt/cloud 可暂停中继，
