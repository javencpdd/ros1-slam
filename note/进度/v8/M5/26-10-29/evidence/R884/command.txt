# 平地增益筛选（3 s，0.3 m/s，heading hold）
for g in "1.0 .5 .4" "2.0 1.0 .5" "0.5 .25 .3"; do
  set -- $g
  python3 -B /tmp/mj_dreamwaq_walk.py --policy /tmp/lite3_walk_policy.onnx \
    --xml /tmp/mjcf/Lite3.xml --mode walk --seconds 3 --v-cmd .3 \
    --heading-hold --yaw-gain $1 --y-gain $2 --max-wz $3 \
    --out /tmp/R884_walk_$1_$2.json
done

# 选定 2.0/1.0/0.5，跑 3 级台阶（6/10/15 cm，深 .5 m，宽 3 m，10 s）
for h in .06 .10 .15; do
  python3 -B /tmp/mj_dreamwaq_walk.py --policy /tmp/lite3_walk_policy.onnx \
    --xml /tmp/mjcf/Lite3.xml --mode stairs --seconds 10 --v-cmd .3 \
    --step-h $h --steps 3 --step-depth .5 --step-width 3.0 --step-front 1.0 \
    --heading-hold --yaw-gain 2.0 --y-gain 1.0 --max-wz .5 \
    --out /tmp/R884_stairs_$h.json
done
