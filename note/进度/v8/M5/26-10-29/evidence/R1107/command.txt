# 0) R1064 负项背景：四例同 lab 在第三例后挂起；逐例单 lab 全过。
#    本轮先在当前 HEAD 连跑 5 次四例同 lab，每次独立端口/输出目录。
#    端口 11990/11992/11994/11996/11998；外层每跑最多 130 s，正常 62—65 s。
for i in 0 1 2 3 4; do
  PORT=$((11990 + i * 2)); OUT=/tmp/r1107_multi_$i
  ROS_MASTER_URI=http://127.0.0.1:$PORT ROS_IP=127.0.0.1 \
  timeout 130 python3 -B regression/runtime_lab.py --out "$OUT" --port "$PORT" \
    --gazebo --local-gazebo \
    --local-cases arrival,localization,sensor_freshness,sensed_no_route
done
