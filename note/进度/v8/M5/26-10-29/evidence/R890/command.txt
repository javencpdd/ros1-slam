# 1) xacro 双态（容器内）
source /opt/ros/noetic/setup.bash
source /opt/ros1-slam/ws/devel/setup.bash
xacro /opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/urdf/lite3_planar.urdf.xacro \
  carrier_visual:=false > /tmp/planar_noviz2.urdf
xacro /opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/urdf/lite3_planar.urdf.xacro \
  carrier_visual:=true > /tmp/planar_viz2.urdf
python3 - <<PY
import xml.etree.ElementTree as ET
for p in ["/tmp/planar_noviz2.urdf","/tmp/planar_viz2.urdf"]:
    r=ET.parse(p).getroot(); print(p, "visuals", len(r.findall(".//visual")))
PY
# 输出：/tmp/planar_noviz2.urdf visuals 0；/tmp/planar_viz2.urdf visuals 2

# 3) 截图用临时世界（只加 GUI 相机；不改仓库世界）
mkdir -p /tmp/r890_payload/stairstudy/worlds
ln -s /opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/urdf /tmp/r890_payload/urdf
cp /opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/stairstudy/worlds/stair_lab_stairs_easy.world \
   /tmp/r890_payload/stairstudy/worlds/stair_lab_stairs_easy.world
# 在 <world name="stair_lab"> 后插入：
# <gui fullscreen="0"><camera name="user_camera">
#   <pose>1.70 -1.70 1.00 0 0.30 2.20</pose>
# </camera></gui>

# 载体图（官方物理载体可见；stairstudy.launch 默认 carrier_visual=true）
roslaunch lite3_gazebo_sim stairstudy.launch gui:=true \
  payload_dir:=/tmp/r890_payload world_file:=stair_lab_stairs_easy.world spawn_x:=0.5
# 模型：ground_plane, stairs_easy, lite3_planar

# 视觉狗图（白盒载体隐藏；stairstudy_visual.launch 默认 carrier_visual=false）
roslaunch lite3_gazebo_sim stairstudy_visual.launch gui:=true \
  payload_dir:=/tmp/r890_payload world_file:=stair_lab_stairs_easy.world spawn_x:=0.5
# 模型：ground_plane, stairs_easy, lite3_planar, lite3_visual
# lite3_visual pose z≈0.33；get_world_properties 四枚在位
