from html.parser import HTMLParser
class P(HTMLParser):
    def __init__(self): super().__init__(); self.stack=[]; self.errors=[]
    def handle_starttag(self,tag,attrs):
        if tag not in ('img','br','meta','link','input','hr','source'): self.stack.append(tag)
    def handle_endtag(self,tag):
        if not self.stack: self.errors.append('extra '+tag); return
        t=self.stack.pop()
        if t!=tag: self.errors.append(f'{t}!={tag}')
p=P(); p.feed(open('web/index.html',encoding='utf-8').read())

python3 - <<'PY'
from html.parser import HTMLParser
from pathlib import Path
root=Path('web'); s=(root/'index.html').read_text(encoding='utf-8')
class P(HTMLParser):
    def __init__(self): super().__init__(); self.ids=set(); self.hrefs=[]; self.srcs=[]
    def handle_starttag(self,tag,attrs):
        d=dict(attrs)
        if 'id' in d: self.ids.add(d['id'])
        if tag=='a' and 'href' in d: self.hrefs.append(d['href'])
        if tag in ('img','script') and 'src' in d: self.srcs.append(d['src'])
        if tag=='link' and 'href' in d: self.srcs.append(d['href'])
p=P(); p.feed(s); miss=[]
for h in p.hrefs:
    if h.startswith('#'):
        if h[1:] and h[1:] not in p.ids: miss.append(('anchor',h))
    elif h.startswith(('http:','https:','mailto:','javascript:')) or h=='': continue
    elif not (root/h.split('#')[0].split('?')[0]).resolve().exists(): miss.append(('href',h))
for x in p.srcs:
    if x.startswith(('http:','https:','data:')): continue
    if not (root/x.split('?')[0]).resolve().exists(): miss.append(('src',x))
print('hrefs',len(p.hrefs),'srcs',len(p.srcs),'missing',miss)
PY
