# 2) 路径空隙筛选（实时 LiDAR；不是物理碰撞证明）

  source /opt/ros/noetic/setup.bash
  source /opt/ros1-slam/ws/devel/setup.bash
  export ROS_MASTER_URI=http://127.0.0.1:11413
  export GAZEBO_MASTER_URI=http://127.0.0.1:11445
  cd /opt/ros1-slam/ws/src/90_platform/v8_platform/framework
  C=/opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/config
  # 2a) R1125 大厅小环：min 0.168 m，9/41 < 0.35 -> 不合格
  python3 -B path_clearance_live.py --path $C/demo_hospital_path.json \
    --sample-m 0.2 --radius .35 --out /tmp/clearance_a.json
  # 2b) 三条已入库安全走廊：min 0.52 m 左右，0 个 < 0.30

      --sample-m 0.2 --radius .30 --out /tmp/clear_safe_$s.json
  done'
# 2c) 其它候选（/tmp，未保留）：x_m8 min .338（1 个 < .35）、x_m6 .130、
#     x_m10 .135、y_n12_n6 .098、loop_m8 .080、loop_m10 .080；均不选。

# 3) 三次 seed（同走廊、不同起止段；8 m、0.2 m/s、40 s、20 Hz）
docker exec ros1-slam-sim-viz bash -lc '
  source /opt/ros/noetic/setup.bash
  source /opt/ros1-slam/ws/devel/setup.bash
  export ROS_MASTER_URI=http://127.0.0.1:11413
  export GAZEBO_MASTER_URI=http://127.0.0.1:11445
  cd /opt/ros1-slam/ws/src/90_platform/v8_platform/framework
  C=/opt/ros1-slam/ws/src/95_simulation/lite3_gazebo_sim/config
  for s in a b c; do
    timeout 150 python3 -B demo_slam_survey.py \
      --path $C/demo_hospital_path_safe_$s.json \
      --duration-s 40 --rate-hz 20 --buffer 5000 \
      --rpe-window-s 1.0 --xy-threshold-m 0.5 \
      --scope demo_hospital_safe_$s --launch \
      --out /tmp/demo_hospital_safe_$s.json
    cp -a /opt/ros1-slam/data/maps/lite3.pcd /tmp/hospital_safe_$s.pcd
  done'
# 每个 JSON 自带 compare / rpe / tracking；RPE 为 1 s 时间窗，tracking 的
# xy 超阈段按首帧平移对齐后统计（避免把 LIO 自有 map 起点偏移算成丢跟踪）。

# 4) PCD 统计（点数、bbox、z 分层、0.5 m XY 覆盖、密度）
