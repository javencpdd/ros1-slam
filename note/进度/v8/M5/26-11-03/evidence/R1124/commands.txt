# 2) Load in Gazebo 11.15.1 (headless, isolated master; did not touch 11313/11345)
docker exec -d ros1-slam-sim-viz bash -lc '
  source /opt/ros/noetic/setup.bash
  export ROS_MASTER_URI=http://127.0.0.1:11413
  export GAZEBO_MASTER_URI=http://127.0.0.1:11455
  export GAZEBO_MODEL_PATH=/opt/ros1-slam/data/imports/aws_hospital_probe/models:/opt/ros1-slam/data/imports/aws_hospital_probe/fuel_models:/usr/share/gazebo-11/models
  export GAZEBO_RESOURCE_PATH=/usr/share/gazebo-11:/opt/ros1-slam/data/imports/aws_hospital_probe
  export GAZEBO_PLUGIN_PATH=/opt/ros1-slam/ws/devel/lib:/opt/ros/noetic/lib
  exec gzserver /opt/ros1-slam/data/imports/aws_hospital_probe/worlds/hospital_two_floors.world \
    -s libgazebo_ros_api_plugin.so --verbose \
    > /opt/ros1-slam/data/imports/aws_hospital_probe/load_two_floors.log 2>&1'
# The same command was run for hospital.world and hospital_three_floors.world.
# A separate isolated roscore was started on 11413 for the ROS API plugin:
#   roscore -p 11413

# 4) Stop by exact PID (do not use a broad pkill -f pattern: it can match the
#    docker exec shell itself and kill the calling command)
docker exec ros1-slam-sim-viz bash -lc '
  p1=$(pgrep -f "^gzserver .*hospital_two_floors" | head -1); [ -n "$p1" ] && kill -TERM "$p1"
  p2=$(pgrep -f "^/usr/bin/python3 /opt/ros/noetic/bin/roscore -p 11413" | head -1); [ -n "$p2" ] && kill -TERM "$p2"'
