# ── B. 有效：世界台阶 + rear_force_bridge 包装 ────────────────
cat > /work/force_world_stair2.py <<'PY'
import json, sys
sys.path.insert(0, '/repo/ws/src/95_simulation/lite3_gazebo_sim/dynamics')
import crawl_gait as crawl
from force_crawl import rear_force_bridge
kw=dict(xml='/work/Lite3_step06.xml', start_x=.80, step_x=1.0, step_h=.06, step_swing=.14,
        drive_fwd=.05, swing_fwd=.06, margin_min=.05, climb_x=.10, climb_dz=.03,
        rear_rounds=4, rear_swing=.08, climb2_x=.06, climb2_dz=.02,
        kp=150., kd=1., swing_kp=100., swing_kd=1.)
with rear_force_bridge() as exp:
    r=crawl.run('stair', **kw)
r['force_reports']=exp['reports']
json.dump(r, open('/work/force_world_stair06b.json','w'), ensure_ascii=False, indent=1)
print(json.dumps({'pass':r.get('pass'),'fell':r.get('fell'),'completed':r.get('completed_steps'),
                  'final_x':r.get('final_x'),'final_z':r.get('final_z'),
                  'force_reports':len(exp['reports'])}))
PY
