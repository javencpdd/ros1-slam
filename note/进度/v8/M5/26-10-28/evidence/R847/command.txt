import json, math
src='note/进度/v8/M5/26-10-28/evidence/R841/a5_gazebo_full.json'
d=json.load(open(src)); rows=d['raw']['physics']; by={}
for r in rows: by.setdefault(r['phase'], []).append(r)
for c in d['cases']:
    if 'trigger' not in c: continue
    ph=c['trigger']['phase']; tr=c['trigger']['wall']; stop=tr+c['trigger_to_physical_stop_wall_s']
    seg=sorted([r for r in by.get(ph,[]) if r['wall']>=tr], key=lambda r:r['wall'])
    path=0.; prev=None; at=None
    for r in seg:
        if r['wall']>stop+1e-6: break
        if prev is not None: path+=math.dist(prev['xyz'][:2], r['xyz'][:2])
        prev=r; at=r
    print(c['case'], 'wall', round(c['trigger_to_physical_stop_wall_s'],4),
          'sim', round(c['trigger_to_physical_stop_sim_s'],4), 'dist', round(path,6))
PY
