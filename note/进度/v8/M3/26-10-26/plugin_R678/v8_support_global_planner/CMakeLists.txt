cmake_minimum_required(VERSION 3.0.2)
project(v8_support_global_planner)

include_directories(include ${catkin_INCLUDE_DIRS})
add_library(v8_support_global_planner src/plugin.cpp)
target_link_libraries(v8_support_global_planner ${catkin_LIBRARIES})
