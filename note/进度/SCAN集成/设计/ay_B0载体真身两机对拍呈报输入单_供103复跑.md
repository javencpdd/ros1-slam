`/tmp/analyze.py`——按 `fields` 的 offset 构 dtype（★Noetic 无 `read_points_numpy`，别照抄 ROS2 写法）：

```python
import rosbag, numpy as np, sys
FMT={1:'i1',2:'u1',3:'i2',4:'u2',5:'i4',6:'u4',7:'f4',8:'f8'}
b=rosbag.Bag(sys.argv[1]); zs=[];rs=[];rings=[];fr=0
for t,m,_ in b.read_messages(topics=["/rslidar_points"]):
    dt=np.dtype({'names':[f.name for f in m.fields],'formats':[FMT[f.datatype] for f in m.fields],
                 'offsets':[f.offset for f in m.fields],'itemsize':m.point_step})
    a=np.frombuffer(m.data,dtype=dt,count=m.width*m.height)
    x,y,z=a['x'].astype(float),a['y'].astype(float),a['z'].astype(float)
    r=np.sqrt(x*x+y*y+z*z); ok=(r>0.05)&np.isfinite(r)&~np.isnan(x)
    rs.append(r[ok]); zs.append(z[ok]); rings.append(a['ring'].astype(int)[ok]); fr+=1
    if fr>=60: break
r=np.concatenate(rs); z=np.concatenate(zs); ring=np.concatenate(rings)
zen=np.degrees(np.arccos(np.clip(z/r,-1,1))); elev=90.0-zen
print("帧=%d 有效点=%d"%(fr,r.size))
print("ring 唯一枚数=%d 取值域=[%d,%d]"%(len(np.unique(ring)),ring.min(),ring.max()))
m2=np.abs(elev)<=15.5
print("|仰角|<=15.5 带内：点数=%d 占%.1f%% ring 枚数=%d"%(m2.sum(),100.0*m2.sum()/r.size,len(np.unique(ring[m2]))))
print("仰角域=[%+.2f,%+.2f]"%(elev.min(),elev.max()))
for d in (0.5,1.0):
    m3=r<d; print("r<%.1fm 点数=%d 占%.2f%%"%(d,m3.sum(),100.0*m3.sum()/r.size))
```
