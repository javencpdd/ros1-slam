import os
RAW = os.environ['RAWD']
CSV = os.path.join(RAW, 'r496_活体窗带真读复读机与lethal独立来源与同名位带件名', 'run', 'csv')
ANCHOR = -50.0
TCOL, TROW = (476, 534), (523, 543)
KS = [(c, r) for c in range(TCOL[0], TCOL[1] + 1) for r in range(TROW[0], TROW[1] + 1)]
S = {}
for t in ('REF', 'A', 'Ap', 'C'):
    with open(os.path.join(CSV, 'cm_%s.csv' % t), encoding='utf-8', errors='replace') as f:
        m = f.readline().strip().split(','); d = [int(v) for v in f.readline().strip().split(',')]
    W, H, res, ox, oy = int(m[1]), int(m[2]), float(m[3]), float(m[4]), float(m[5])
    S[t] = dict(W=W, H=H, res=res, ox=ox, oy=oy, d=d,
                sh=(int(round((ox - ANCHOR) / res)), int(round((oy - ANCHOR) / res))))
    print('%s META＝%s 锚平移=(%+d,%+d)' % (t, ','.join(m), S[t]['sh'][0], S[t]['sh'][1]))
def val(t, c, r):
    g = S[t]; fc, fr = c - g['sh'][0], r - g['sh'][1]
    if not (0 <= fc < g['W'] and 0 <= fr < g['H']): return None
    return g['d'][fr * g['W'] + fc]
def raw(t, c, r):
    g = S[t]
    return g['d'][r * g['W'] + c]
PRED = dict(P100=lambda v: v == 100, P90=lambda v: v is not None and v >= 90)
print()
print('== 〔一〕跨态：同下标硬比（整份 1e6 平坦）‖ 世界对齐（锚架 1e6）‖ 盲区（任一态越界）==')
for a, b in (('A','Ap'), ('A','C'), ('A','REF'), ('REF','A')):
    ga, gb = S[a], S[b]
    nraw = sum(1 for i in range(len(ga['d'])) if ga['d'][i] != gb['d'][i])
    ali = sum(1 for c in range(1000) for r in range(1000)

    blind = sum(1 for c in range(1000) for r in range(1000) if val(a, c, r) is None or val(b, c, r) is None)
    same = (ga['ox'] == gb['ox'] and ga['oy'] == gb['oy'])
    print('%s<->%s origin同=%s raw=%d aligned=%d 盲区=%d 不变量(同origin时raw==aligned)=%s'
          % (a, b, same, nraw, ali, blind, (nraw == ali) if same else '不适用'))
print()
print('== 〔四〕T 域逐格差（aligned 与 raw 两口径）==')
for p in ('P100', 'P90'):
