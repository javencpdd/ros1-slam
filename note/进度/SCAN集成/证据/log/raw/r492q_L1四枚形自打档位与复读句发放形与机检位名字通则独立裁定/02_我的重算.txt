    return {"w":int(m[1]),"h":int(m[2]),"res":float(m[3]),"ox":float(m[4]),"oy":float(m[5]),"d":d}
G={t:load(t) for t in ("A","Ap","C","REF")}
for t in ("A","Ap","C","REF"):
    g=G[t]; print("%s|META现读 w=%d h=%d res=%.4f origin=(%.4f,%.4f) len(data)=%d"%(t,g["w"],g["h"],g["res"],g["ox"],g["oy"],len(g["d"])))
def sh(t):
    g=G[t]; return (int(round((g["ox"]-AN)/g["res"])), int(round((g["oy"]-AN)/g["res"])))
def val(t,c,r):
    g=G[t]; sc,sr=sh(t); fc,fr=c-sc,r-sr
    return None if not(0<=fc<g["w"] and 0<=fr<g["h"]) else g["d"][fr*g["w"]+fc]
def raw(t,c,r):
